table_lum_CarTrans_overall <- table_lum_CarTrans_overall %>%
  mutate(
    Visit = as.character(Visit),
    Visit = case_when(
    Visit == "Healthy" ~ "Healthy",
    Visit == "V0" ~ "Baseline",
    Visit == "V1" ~ "Week 1",
    Visit == "V2" ~ "Week 2",
    Visit == "V3" ~ "Month 1-3",
    Visit == "V4" ~ "Month 4-6",
    Visit == "V5" ~ "Month 12",
    grepl("^VX", Visit) ~ Visit, #keep different X timepoints to not create duplicates
    TRUE~Visit
    ),
    Visit = factor(Visit,levels = c("Healthy","Baseline","Week 1","Week 2","Month 1-3","Month 4-6","Month 12", "VX1", "VX2", "VX3", "VX4")) 
  )

table_convalesence <- table_lum_CarTrans_overall %>%filter(Visit %in% c("Healthy", "Month 1-3")) %>%
  droplevels()

# Calculate p-values between Healthy and Month 1-3
     # Create the loop.vector (all the parameter columns)
    names(which(sapply(table_convalesence, is.numeric) == TRUE)) -> parameters 

  # Create table for stats
    table_convalesence_stat <- data.table (Parameter = numeric(), group1 = character(), 
                                              group2 = character(), n1 = numeric(),
                                              n2 = numeric(), statistic = numeric (),
                                              p = numeric (), p.signif = character(), FDR = numeric()) #FDR False Discovery Rate
   
  # Create loop for calculation of stat test by parameter between Healthy and Month 1-3
    for (i in parameters)
      
    { 
      # Formula to insert parameter in loop 
        formula = as.formula( paste(i, "Visit", sep="~") )
          
        stat_parameter <- table_convalesence %>% 
                          wilcox_test(formula = formula) %>%
                          add_significance() %>%
                          mutate(FDR = p.adjust(p, method = "BH")) %>% #Benjamini-Hochberg correction
                          rename(Parameter = .y.)
        
      # Combine currently calculated stats with previous calculations
        table_convalesence_stat <- rbind(table_convalesence_stat, stat_parameter)
    }

# Calculate FC between different Visits
    
      # Calculate Mean of parameters per group
        table_convalesence_log2  <- table_convalesence %>% 
                                       group_by(Visit) %>%
                                       summarise(across(CCL11:Grem2, ~ mean(.x, na.rm = TRUE)))
      
      # Transverse Data Table and set rownames
        table_convalesence_log2_t  <- table_convalesence_log2  %>% 
                                         t %>% 
                                         as.data.frame() %>% 
                                         row_to_names(1)
     
     # Convert whole data table to numeric
       table_convalesence_log2_t  <- mutate_all(table_convalesence_log2_t, 
                                                   function(x) as.numeric(as.character(x)))
       
     # Calculate FC between Healthy and Month 1-3
       table_convalesence_log2_t$log2_fc <- log2(table_convalesence_log2_t$Healthy /
                                                    table_convalesence_log2_t$`Month 1-3`)
       
     # Add p-values to data table
        # Extract only Parameters and p-values
           stat.test_p <- table_convalesence_stat %>% 
                          dplyr::select(Parameter, FDR)

        # Merge data tables (log2, FC, p val)
          table_convalesence_log2_t$Parameter <- rownames(table_convalesence_log2_t)
          table_convalesence_volcano <- merge(table_convalesence_log2_t, stat.test_p, by = "Parameter")
 
  # Create Volcano Plot
          
     # Mark parameters if FC < -1.5 = downregulated
       table_convalesence_volcano$diff_expression <- ifelse(table_convalesence_volcano$log2_fc < -1.5, "down",
                                                            ifelse(table_convalesence_volcano$log2_fc > 1.5, "up", "no"))
     
     # Add new column to relabel parameters FC of <-1.5
      table_convalesence_volcano$label_para <- ifelse(table_convalesence_volcano$log2_fc < -1.5 |
                                                        table_convalesence_volcano$log2_fc > 1.5,
                                                      table_convalesence_volcano$Parameter, "")   
    
      volcano_convalescence <- EnhancedVolcano(table_convalesence_volcano, 
                                      lab = table_convalesence_volcano$label_para,
                                      x = 'log2_fc', y = 'FDR', 
                                      FCcutoff = 1.5, pointSize = 3.5, labSize = 4, colAlpha = 0.7,

                                      drawConnectors = TRUE,widthConnectors = 0.5, colConnectors = "darkred") +
