# Exclude Healthy Condition from the comparison
table_lum_CarTrans_overall <- table_lum_CarTrans_overall %>%filter(Visit != c("Healthy", "Baseline"))

# Rename Healthy vs all CarTrans Visits
  table_lum_CarTrans_overall$ISHLT_2004 <- ifelse(table_lum_CarTrans_overall$ISHLT_2004 %in% c("ACR 0", "ACR 1R"), "NoRejection", "Rejection")

# Calculate p-values between Rejection and No Rejection
    
  # Create the loop.vector (all the parameter columns)
    names(which(sapply(table_lum_CarTrans_overall, is.numeric) == TRUE)) -> parameters 

  # Create table for stats
    table_lum_CarTrans_overall_stat <- data.table (Parameter = numeric(), group1 = character(), 
                                              group2 = character(), n1 = numeric(),
                                              n2 = numeric(), statistic = numeric (),
                                              p = numeric (), p.signif = character(), FDR = numeric()) #FDR False Discovery Rate
   
  # Create loop for calculation of stat test by parameter between Rejection and No Rejection
    for (i in parameters)
      
    { 
      # Formula to insert parameter in loop 
        formula = as.formula( paste(i, "ISHLT_2004", sep="~") )
          
        stat_parameter <- table_lum_CarTrans_overall %>% 
                          wilcox_test(formula = formula) %>%
                          add_significance() %>%
                          mutate(FDR = p.adjust(p, method = "BH")) %>% #Benjamini-Hochberg correction
                          rename(Parameter = .y.)
        
      # Combine currently calculated stats with previous calculations
        table_lum_CarTrans_overall_stat <- rbind(table_lum_CarTrans_overall_stat, stat_parameter)
    }

# Calculate FC between different Visits
    
      # Calculate Mean of parameters per group
        table_lum_CarTrans_overall_log2  <- table_lum_CarTrans_overall %>% 
                                       group_by(ISHLT_2004) %>%
                                       summarise(across(CCL11:BMP4, ~ mean(.x, na.rm = TRUE)))
      
      # Transverse Data Table and set rownames
        table_lum_CarTrans_overall_log2_t  <- table_lum_CarTrans_overall_log2  %>% 
                                         t %>% 
                                         as.data.frame() %>%
