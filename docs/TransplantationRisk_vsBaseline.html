Transplantation risk on overall analytes (z-score)

#Define analytes
analyte_cols <- setdiff(colnames(table_lum_CarTrans_overall), c("Study_ID", "Timepoint", "Sample_ID", "ISHLT_2004", "Visit", "Transplantation_risk", "Class_Sup"))

# Log transform input values
table_lum_CarTrans_overall_z <- table_lum_CarTrans_overall %>%
  mutate(across(all_of(analyte_cols), ~ log2(pmax(., 1))))

# Calculate healthy mean and SD as reference
BL_ref <- table_lum_CarTrans_overall_z %>%
  filter(Visit == "Baseline") %>%
  summarise(across(all_of(analyte_cols),
                   list(mean = ~mean(., na.rm = TRUE),
                        sd   = ~sd(., na.rm = TRUE))))

# Calculate Z score
z_data <- table_lum_CarTrans_overall_z

for (col in analyte_cols) {
  mu <- BL_ref[[paste0(col, "_mean")]]
  sd <- BL_ref[[paste0(col, "_sd")]]
  
  if (is.na(sd) || sd == 0) {
    z_data[[col]] <- NA
  } else {
    z_data[[col]] <- (z_data[[col]] - mu) / sd
  }
}

# Set global score per sample
z_data <- z_data %>%mutate(global_z = rowMeans(across(all_of(analyte_cols)), na.rm = TRUE))
z_data_plot <- z_data %>%
  filter(!Timepoint %in% c("Baseline")) %>%
  mutate(
    Transplantation_risk = factor(Transplantation_risk),
    Timepoint = factor(Timepoint,
                        levels = c("Baseline", "Week 1", "Week 2",
                                   "Month 1-3", "Month 4-6", "Month 12",
                                   "Rejection Context"))
