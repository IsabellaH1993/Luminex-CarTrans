Vergleich vs Baseline: same set of patients- paired wilcox test used

res_vs_baseline <- df_pdl1 %>%
  filter(Timepoint != "Healthy") %>%
  group_by(Study_ID, Timepoint) %>%
  summarise(PD_L1 = mean(PD_L1)) %>%
  pivot_wider(names_from = Timepoint, values_from = PD_L1) %>%
  pivot_longer(-c(Study_ID, Baseline), names_to = "Timepoint", values_to = "PD_L1") %>%
  filter(!is.na(Baseline), !is.na(PD_L1)) %>%
  mutate(Timepoint = factor(Timepoint, levels = levels(df_pdl1$Timepoint))) %>%
  group_by(Timepoint) %>%
  summarise(n_pairs = n(),
            median_change = median(PD_L1 - Baseline),
            p = wilcox.test(PD_L1, Baseline, paired = TRUE, exact = FALSE)$p.value) %>%
  adjust_pvalue(method = "BH") %>%
  add_significance("p.adj")

res_followups <- df_pdl1 %>%
  filter(!Timepoint %in% c("Healthy", "Baseline")) %>%
  group_by(Study_ID, Timepoint) %>%
  summarise(PD_L1 = mean(PD_L1), .groups = "drop") %>%
  {inner_join(., ., by = "Study_ID", suffix = c(".1", ".2"), relationship = "many-to-many")} %>%
  filter(as.integer(Timepoint.1) < as.integer(Timepoint.2)) %>%
  group_by(group1 = as.character(Timepoint.1), group2 = as.character(Timepoint.2)) %>%
  summarise(n_pairs = n(),
            median_change = median(PD_L1.2 - PD_L1.1),
            p = wilcox.test(PD_L1.2, PD_L1.1, paired = TRUE, exact = FALSE)$p.value,
            .groups = "drop") %>%
  adjust_pvalue(method = "BH") %>%
  add_significance("p.adj")
