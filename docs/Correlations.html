# Define lab parameter (y axis)
  lab_param <- "NTproBNP"

  analytes <- c("PD_L1", "CXCL10", "IL2R", "Grem2")

  plots_list <- list()

# Loop Correlation Plot
  for (p in analytes) 
    
    {
  
      # Compute Spearman correlation
        cor_test <- cor.test(table_corr[[p]], table_corr[[lab_param]], method = "spearman")
  
      # Define x axis margins and breaks
        max_x <- max(table_corr[[p]], na.rm = TRUE) * 1.1
        min_x <- min(table_corr[[p]], na.rm = TRUE)
        max_log10_limit <- 10^(ceiling(log10(max_x)))
        min_log10_limit <- 10^(floor(log10(min_x)))
        x_breaks <- 10^(seq(log10(min_log10_limit), log10(max_log10_limit), by = 1))
  
      # Scatter plot with regression line
        plot <- ggplot(table_corr, aes(x = .data[[p]], y = .data[[lab_param]])) +
                geom_point(aes(fill = Timepoint), shape = 21, size = 2.5, color = "black",
                           position = position_jitter(width = 0.05, height = 0)) +
                geom_smooth(method = "lm", se = TRUE, color = "black") +
                scale_fill_manual(values = ann_colors$Timepoint) +
                scale_x_log10(labels = function(y) format(y, scientific = FALSE, trim = TRUE), 
                              limits = c(min_log10_limit, max_log10_limit),
                              expand = c(0.05, 0), breaks = x_breaks) +
                theme_classic() +
                labs(x = analyte_labels[[p]], y = lab_label) +
                annotate("text", x = -Inf, y = Inf,
                         label = paste0("r = ", round(cor_test$estimate, 2),
                                        "\np = ", ifelse(cor_test$p.value < 0.001, "< 0.001", signif(cor_test$p.value, 3))),
                         hjust = -0.1, vjust = 1.2, size = 4)
        
      # Store each plot
        plots_list[[p]] <- plot

  }

# Combine plots to a panel
  panel <- ggarrange(plotlist = plots_list, ncol = 2, nrow = 2,
                     common.legend = TRUE, legend = "right")
