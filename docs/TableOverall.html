# Mutate factors and define different levels

## Clinical Data
data$`Clinical Data` <- data$`Clinical Data` |>
  dplyr::mutate(

    ### Short-term MCS
    Shortterm_MCS = dplyr::case_when(
      Shortterm_MCS %in% c("No", "NA.") ~ "No",
      Shortterm_MCS %in% c("VA-ECMO, Impella") ~ "Impella",
      Shortterm_MCS %in% c("VA-ECMO", "postoperative VA-ECMO") ~ "ECMO",
      TRUE ~ NA_character_
    ),

    ### Long-term MCS
    Longterm_MCS = dplyr::case_when(
      Longterm_MCS %in% c("No", "NA") ~ "No",
      Longterm_MCS == "LVAD" ~ "LVAD",
      TRUE ~ NA_character_
    ),

    ### Sex
    Sex = dplyr::case_when(
      Sex == "M" ~ "Male",
      Sex == "F" ~ "Female",
      TRUE ~ NA_character_
    ),
    
    ### Diabetes
    Diabetes = dplyr::case_when(
      Diabetes %in% c("No", "prediabetes") ~ "No",
      Diabetes %in% c("DM type 2") ~ "Yes",
      TRUE ~ NA_character_
    ),

    ### Device
    Device = dplyr::case_when(
      Device %in% c("No", "NA") ~ "No", 
      Device %in% c( "CRT-D", "CRT-D, VSD Patch Plastic, aortic valve mechanic prosthetic",
                    "St.n CRT-D") ~ "CRT-D",
      Device %in% c("ICD", "S-ICD", "ICD-D") ~ "ICD",
      Device == "life vest" ~ "Life Vest",
      TRUE ~ NA_character_
    )
  ) |>
  dplyr::mutate(
    Shortterm_MCS = factor(Shortterm_MCS, levels = c("No", "Impella", "ECMO")),
    Longterm_MCS  = factor(Longterm_MCS,  levels = c("No", "LVAD")),
    Sex           = factor(Sex,           levels = c("Female", "Male")),
    Device        = factor(Device,        levels = c("No", "ICD", "CRT-D", "Life Vest"))
  )
